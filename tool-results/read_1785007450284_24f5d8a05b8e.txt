   298→function SeverityBadge({ severity }: { severity: string }) {
   299→  const config = {
   300→    high: { label: 'High Impact', className: 'bg-red-500/15 text-red-600 dark:text-red-400 border-red-500/20' },
   301→    medium: { label: 'Medium Impact', className: 'bg-amber-500/15 text-amber-600 dark:text-amber-400 border-amber-500/20' },
   302→    low: { label: 'Low Impact', className: 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400 border-emerald-500/20' },
   303→  };
   304→  const c = config[severity] || config.low;
   305→  return (
   306→    <span className={`inline-flex items-center rounded-full border px-2.5 py-0.5 text-[11px] font-semibold ${c.className}`}>
   307→      {c.label}
   308→    </span>
   309→  );
   310→}
